import numpy as np import matplotlib.pyplot as plt from sklearn.datasets import make_blobs from sklearn.cluster import KMeans # Number of samples n_samples = 300 # Number of clusters n_clusters = 4 # Generate sample data x, y = make_blobs( n_samples=n_samples, centers=n_clusters, random_state=42 ) # Create K-Means model kmeans = KMeans( n_clusters=n_clusters, random_state=42, n_init=10 ) # Train the model kmeans.fit(x) # Predict cluster labels y_kmeans = kmeans.predict(x) # Create plot plt.figure(figsize=(8, 6)) # Plot data points plt.scatter( x[:, 0], x[:, 1], c=y_kmeans, s=50, cmap='viridis' ) # Get cluster centers centers = kmeans.cluster_centers_ # Plot cluster centers plt.scatter( centers[:, 0], centers[:, 1], c='red', s=200, alpha=0.5, marker='x' ) # Graph details plt.title('K-Means Clustering') plt.xlabel('Feature 1') plt.ylabel('Feature 2') # Display graph plt.show()
